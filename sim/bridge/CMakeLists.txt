cmake_minimum_required(VERSION 3.16)
project(drone137_image_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(JPEG REQUIRED)
find_package(Threads REQUIRED)
add_executable(drone137-image-bridge main.cpp)
target_link_libraries(drone137-image-bridge PRIVATE gz-transport13::gz-transport13 gz-msgs10::gz-msgs10 JPEG::JPEG Threads::Threads)
add_executable(drone137-pose-bridge pose.cpp)
target_link_libraries(drone137-pose-bridge PRIVATE gz-transport13::gz-transport13 gz-msgs10::gz-msgs10 Threads::Threads)
